```python
"""角色业务（REQ-20261020-01）：等级判定 + 超管启动提权。

被 api/deps（require_role）与 api/v1/users（角色调整）、main.lifespan（提权）复用。
"""

import logging

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.user import (
    ROLE_ADMIN,
    ROLE_CUSTOMER,
    ROLE_LEVEL,
    ROLE_PENDING,
    ROLE_SUPER_ADMIN,
    User,
)

logger = logging.getLogger("app")

#: 接口可赋的角色——super_admin 只能由 SUPER_ADMIN_EMAIL 启动机制产生（REQ §1.3）
ASSIGNABLE_ROLES = (ROLE_ADMIN, ROLE_CUSTOMER, ROLE_PENDING)


def can_assign_role(operator: User, target: User, new_role: str) -> bool:
    """等级规则（REQ §1.3）：不许动自己；只能动严格低于自己的人；新角色不得超过自己等级。"""
    if operator.id == target.id:
        return False
    if new_role not in ASSIGNABLE_ROLES:
        return False
    return (
        ROLE_LEVEL[operator.role] > ROLE_LEVEL[target.role]
        and ROLE_LEVEL[new_role] <= ROLE_LEVEL[operator.role]
    )


def ensure_super_admin(db: Session, email: str) -> User | None:
    """启动幂等提权（REQ §1.4）：账号存在则确保 super_admin，缺失记日志跳过。

    不 bump credential_version——超管本人无需重新登录。
    """
    normalized = email.strip().lower()
    if not normalized:
        return None
    user = db.scalar(select(User).where(User.email == normalized))
    if user is None:
        logger.warning("SUPER_ADMIN_EMAIL=%s 未注册，跳过超管提权", normalized)
        return None
    if user.role != ROLE_SUPER_ADMIN:
        user.role = ROLE_SUPER_ADMIN
        db.commit()
        logger.info("用户 %s 已提权为 super_admin", normalized)
    return user
```

def require_role(*allowed: str):
    """角色门禁（REQ-20261020-01 §1.5）：先过 get_current_user，再校验角色。

    用法：端点 dependencies=[Depends(require_role(ROLE_ADMIN, ROLE_SUPER_ADMIN))]。
    pending 用户任何数据接口即 403；401 语义（未登录）不变。
    """

    def _checker(user: User = Depends(get_current_user)) -> User:
        if user.role not in allowed:
            raise ApiError(403, "FORBIDDEN", "insufficient role")
        return user

    return _checker
```

"""用户管理路由（REQ-20261020-01 §1.3/§1.6，DESIGN §2.2）。

- GET   /users               列表（admin+）
- PATCH /users/{id}/role     角色调整（等级规则在 role_service.can_assign_role）
"""

from fastapi import APIRouter, Depends
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.api.deps import get_current_user, get_db, require_role
from app.api.errors import ApiError
from app.models.user import ROLE_ADMIN, ROLE_SUPER_ADMIN, User
from app.schemas.users import UserRoleUpdateRequest, UserListItem, UsersListResponse
from app.services import role_service
from app.schemas.auth import UserPublic

router = APIRouter(prefix="/users", tags=["users"])


@router.get(
    "",
    response_model=UsersListResponse,
    summary="用户列表（管理员）",
    description="全部账号按注册时间倒序：邮箱/昵称/角色/注册时间/最近登录。"
    "仅 admin 与 super_admin 可访问（REQ-20261020-01）。",
)
def list_users(
    db: Session = Depends(get_db),
    _: User = Depends(require_role(ROLE_ADMIN, ROLE_SUPER_ADMIN)),
) -> UsersListResponse:
    """GET /users。"""
    users = db.scalars(select(User).order_by(User.created_at.desc(), User.id.desc())).all()
    return UsersListResponse(users=[UserListItem.model_validate(u) for u in users])


@router.patch(
    "/{user_id}/role",
    response_model=UserPublic,
    summary="调整用户角色（等级制）",
    description="可赋 admin/customer/pending。规则：只能操作等级严格低于自己的用户，"
    "新角色不得超过自己等级，不能操作自己（REQ-20261020-01 §1.3）。"
    "super_admin 角色不可经此接口赋值。",
)
def set_user_role(
    user_id: int,
    payload: UserRoleUpdateRequest,
    db: Session = Depends(get_db),
    operator: User = Depends(require_role(ROLE_ADMIN, ROLE_SUPER_ADMIN)),
) -> UserPublic:
    """PATCH /users/{id}/role。不 bump 凭据版本：目标会话不中断，下一请求按新角色生效。"""
    target = db.get(User, user_id)
    if target is None:
        raise ApiError(404, "USER_NOT_FOUND", "user not found")
    if not role_service.can_assign_role(operator, target, payload.role):
        raise ApiError(403, "FORBIDDEN", "not allowed to assign this role")
    target.role = payload.role
    db.commit()
    db.refresh(target)
    return UserPublic.model_validate(target)
```

/** 角色归属主页（REQ-20261020-01）：pending→提示页，customer→客户视图，其余→看板 */
function homePathFor(role) {
  if (!role || role === 'pending') return '/pending'
  if (role === 'customer') return '/customer'
  return '/'
}

/** 角色守卫：角色不在 allow 内 → 跳该角色主页（不渲染受限内容） */
function RequireRole({ allow, children }) {
  const { user } = useAuth()
  if (!user) return children // 未登录由外层 RequireAuth 处理
  if (!allow.includes(user.role)) return <Navigate to={homePathFor(user.role)} replace />
  return children
}

/** /pending 专属：非 pending 角色跳自己主页 */
function PendingOnly({ children }) {
  const { user } = useAuth()
  if (user && user.role !== 'pending') return <Navigate to={homePathFor(user.role)} replace />
  return children
}
```

// 与后端 role_service 同源的展示用规则（REQ §1.3）
const LEVEL = { super_admin: 3, admin: 2, customer: 1, pending: 0 }
const ASSIGNABLE = ['admin', 'customer', 'pending']
const ROLE_KEY = {
  super_admin: 'users.roleSuperAdmin',
  admin: 'users.roleAdmin',
  customer: 'users.roleCustomer',
  pending: 'users.rolePending',

  const navigate = useNavigate()
  usePageTitle('common.pageTitle.users')

  const [actionError, setActionError] = useState(false)

    setLoadError(false)

      setLoadError(true)

        {users !== null && users.length > 0 ? (
          <table className="users-table">
            <thead>
              <tr>
                <th>{t('users.colEmail')}</th>
                <th>{t('users.colNickname')}</th>

            </thead>
            <tbody>
              {users.map((row) => (
                <tr key={row.id}>
                  <td>{row.email}</td>
                  <td>{row.nickname}</td>
                  <td>{t(ROLE_KEY[row.role] || 'users.rolePending')}</td>
                  <td>{fmtDateTime(row.created_at)}</td>
                  <td>{fmtDateTime(row.last_login_at)}</td>
                  <td className="users-actions">
                    {actionsFor(row).map((role) => (
                      <button
                        key={role}
                        type="button"
                        className="chip"
                        disabled={busyId === row.id}
                        onClick={() => handleSetRole(row, role)}
                      >
                        {t(ACTION_KEY[role])}
                      </button>
                    ))}
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        ) : null}
      </div>
    </div>
  )
}
```

注：`AppBar` 在本任务内改为 `navs` 数组（见 Step 5），`onRefresh`/`onGuide`/`status` 均为可选 prop，不传即不渲染。加载态那行 `<div className="statusline">` 若 `dashboard.refresh` 文案不合适，可新增 `users.loading` 键（zh「加载中…」/ en "Loading…"）替换——二选一，实现时保持 en/zh 同步即可。
